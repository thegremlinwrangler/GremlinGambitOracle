    { text: "I am Going to Need\nA Moment\nand a Fuckin' biscuit.", tier: "nuclear" },
  ];

  const tierLabels = {
    mild: "Mild Disdain",
    medium: "Full Refusal",
    nuclear: "Nuclear Option"
  };

  let lastIndex = -1;
  let isShaking = false;

  function getAnswer() {
    let idx;
    do { idx = Math.floor(Math.random() * answers.length); }
    while (idx === lastIndex && answers.length > 1);
    lastIndex = idx;
    return answers[idx];
  }

  function shake() {
    if (isShaking) return;
    isShaking = true;

    const answerEl = document.getElementById('answerText');
    const intensityEl = document.getElementById('intensity');
    const wrapper = document.getElementById('wrapper');

    answerEl.classList.remove('visible');
    intensityEl.classList.remove('visible', 'mild', 'medium', 'nuclear');

    wrapper.classList.add('shaking');
    document.body.classList.add('shaking');

    wrapper.addEventListener('animationend', () => {
      wrapper.classList.remove('shaking');
      document.body.classList.remove('shaking');

      const result = getAnswer();
      answerEl.textContent = result.text;
      answerEl.classList.add('visible');

      intensityEl.textContent = tierLabels[result.tier];
      intensityEl.classList.add('visible', result.tier);

      isShaking = false;
    }, { once: true });
  }

  document.body.addEventListener('click', shake);

  let lastShakeTime = 0;
  if (window.DeviceMotionEvent) {
    window.addEventListener('devicemotion', (e) => {
      const acc = e.accelerationIncludingGravity;
      if (!acc) return;
      const total = Math.abs(acc.x) + Math.abs(acc.y) + Math.abs(acc.z);
      const now = Date.now();
      if (total > 25 && now - lastShakeTime > 1000) {
        lastShakeTime = now;
        shake();
      }
    });
  }

  if (typeof DeviceMotionEvent !== 'undefined' &&
      typeof DeviceMotionEvent.requestPermission === 'function') {
    document.body.addEventListener('click', () => {
      DeviceMotionEvent.requestPermission().catch(() => {});
    }, { once: true });
  }
</script>
